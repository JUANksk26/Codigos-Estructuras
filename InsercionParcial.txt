    static long intercambios  = 0;

    public static void insercion(String[] arr) {
        int n = arr.length;
        for (int i = 1; i < n; i++) {
            iteraciones++; 
            String key = arr[i];
            int j = i - 1;
            while (j >= 0) {
                comparaciones++;
                if (arr[j].compareToIgnoreCase(key) > 0) {
                    intercambios++;
                    arr[j + 1] = arr[j];
                    j--;
                } else {
                    break;
                }
            }
            arr[j + 1] = key;
        }
    }
